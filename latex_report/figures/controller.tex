  ]
  % ---------------- cores, private caches, gates
  \foreach \x/\n in {0/Core 0, 2.3/Core 1, 5.6/Core $N\!-\!1$} {
    \node[box, fill=blue!6,  minimum width=17mm, minimum height=6mm] at (\x,0)    {\n};
    \node[box, fill=gray!10, minimum width=17mm, minimum height=7mm] at (\x,-1.1) {L1D + L2\\[-1pt]prefetcher};
    \node[box, fill=red!10,  minimum width=17mm, minimum height=5mm] at (\x,-2.0) {filter};
    \draw[thick] (\x,-0.3) -- (\x,-0.75);
    \draw[thick] (\x,-1.45) -- (\x,-1.75);

  }
  \node at (3.95,0) {$\cdots$};
  \node at (3.95,-1.1) {$\cdots$};
  \node at (3.95,-2.0) {$\cdots$};
  \draw[red!70!black, dashed] (0.85,-2.0) -- (1.45,-2.0);
  \draw[red!70!black, dashed] (3.15,-2.0) -- (4.75,-2.0);

  % ---------------- shared LLC (way-partitioned, split UMON) and DRAM
  \node[box, fill=gray!10, minimum width=80mm, text width=78mm, minimum height=10mm] (llc) at (2.85,-3.3)
        {Shared LLC, way-partitioned (way quota per core)\\[-1pt]
         UMON shadow tags: hits per LRU position split into \textbf{demand} / \textbf{prefetch}};
  \node[box, fill=gray!10, minimum width=80mm, text width=78mm, minimum height=6mm] (dram) at (2.85,-4.6)
        {DRAM (how busy: share of peak bandwidth, 4 levels, per epoch)};
  \draw[thick] (llc.south) -- (dram.north);

  % ---------------- coordinator
  \node[box, fill=orange!12, minimum width=40mm, minimum height=55mm, text width=37mm, align=left,
        anchor=north west] (ctl) at (10.4,0.35) {%
     \centering\textbf{Coordinator}\\[-1pt] (once per epoch, $10^6$ cycles)\\[3pt]
     \raggedright
     \textbf{1}\ DRAM busy? rank cores by useless / way-scaled total prefetch traffic\\[2pt]
     \textbf{2}\ try one change: skip 1 epoch, measure 2\\[2pt]
     \textbf{3}\ keep throttle iff $F\uparrow$, $T\ge -0.5\%$, own core $\ge -1\%$\\[2pt]
     \textbf{4}\ revalidate (8$\to$64 epochs), calm release, phase reset\\[2pt]
     \textbf{5}\ accuracy hold ($<0.15$) with probes\\[2pt]
     \textbf{6}\ every 5 epochs: UCP lookahead on demand $+\,w\cdot$prefetch hits};

  % ---------------- inputs (blue) and actuations (red), one height each
  \draw[sig] (6.45,0)    -- node[lab, above] {IPC per core} (10.4,0);

  \draw[act] (10.4,-2.0)  -- node[lab, above] {throttle level $\ell_i\in\{0..3\}$} (6.45,-2.0);
  \draw[act] (10.4,-3.0)  -- node[lab, above] {way quota per core} (6.85,-3.0);
  \draw[sig] (6.85,-3.6)  -- node[lab, below, pos=0.55] {demand / prefetch hits} (10.4,-3.6);
  \draw[sig] (6.85,-4.6)  -- node[lab, above] {bandwidth level} (10.4,-4.6);

  % ---------------- legend
  \node[lab, anchor=west, font=\tiny] at (-0.85,-5.25) {%
     \textcolor{blue!65!black}{\rule[0.5ex]{4mm}{1pt}} measured inputs \qquad
     \textcolor{red!70!black}{\rule[0.5ex]{4mm}{1pt}} actuations \qquad
     $F$: way-weighted fairness, $T$: throughput};
